**Tech Stack :** Python 3.13, uv, FastAPI, psycopg 3, pytest, ruff, mypy ; Node 22, Vite, Vitest, oxlint,

```js
// node ops/inventaire.mjs ~/PERSO/smart-receipe-scheduler/src > docs/inventaire-surface.md
import { readdirSync, readFileSync, statSync } from 'node:fs'
import { join, relative } from 'node:path'

const racine = process.argv[2]
const fichiers = []
const parcours = d => { for (const f of readdirSync(d)) { const p = join(d, f)
  if (statSync(p).isDirectory()) parcours(p)
  else if (/\.(ts|tsx)$/.test(f) && !/\.test\.tsx?$/.test(f)) fichiers.push(p) } }
parcours(racine)

const lignes = []
for (const f of fichiers) {
  const t = readFileSync(f, 'utf8')
  for (const m of t.matchAll(/\.from\(\s*['"`]([\w-]+)['"`]\s*\)([\s\S]{0,400}?)\.(select|insert|update|upsert|delete)\(/g))
    lignes.push(['table', m[1], m[3], relative(racine, f)])
  for (const m of t.matchAll(/\.rpc\(\s*['"`]([\w-]+)['"`]/g)) lignes.push(['rpc', m[1], 'appel', relative(racine, f)])
  for (const m of t.matchAll(/functions(?:\.invoke\(\s*|\/v1\/)['"`]?([\w-]+)/g)) lignes.push(['fonction', m[1], 'appel', relative(racine, f)])
  if (/\.channel\(/.test(t)) lignes.push(['temps réel', '-', 'abonnement', relative(racine, f)])
  for (const m of t.matchAll(/\.storage\s*\.from\(\s*['"`]([\w-]+)/g)) lignes.push(['stockage', m[1], 'appel', relative(racine, f)])
}
const tri = [...new Map(lignes.map(l => [l.join('|'), l])).values()].sort((a, b) => a.join().localeCompare(b.join()))
console.log('# Inventaire de la surface appelée par le front (avant le socle)\n')
console.log(`Produit par \`ops/inventaire.mjs\` le ${new Date().toISOString().slice(0, 10)}. Une ligne = une sorte d'appel dans un fichier.\n`)
console.log('| Sorte | Nom | Geste | Fichier |\n|---|---|---|---|')
for (const l of tri) console.log(`| ${l.join(' | ')} |`)
const compte = s => new Set(tri.filter(l => l[0] === s).map(l => l[1])).size
console.log(`\n**Totaux** : ${compte('table')} tables, ${compte('rpc')} fonctions SQL, ${compte('fonction')} fonctions Edge, ` +
  `${compte('stockage')} seaux, temps réel dans ${tri.filter(l => l[0] === 'temps réel').length} fichier(s).`)
```

```python
# src/popote_api/errors.py
"""Toutes les erreurs de l'API au format Problem Details (RFC 9457)."""
from http import HTTPStatus
from typing import Any

from fastapi import FastAPI, Request
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse
from starlette.exceptions import HTTPException as StarletteHTTPException

TYPE_DE_BASE = "https://notre-popote.fr/problemes/"
MEDIA = "application/problem+json"
TITRES = {
    400: "Requête mal formée", 401: "Pas de session", 403: "Interdit", 404: "Introuvable",
    405: "Méthode non permise", 409: "Conflit", 422: "Données invalides", 429: "Trop de requêtes",
    500: "Erreur interne", 503: "Service indisponible",
}
CODES = {404: "introuvable", 405: "methode_non_permise", 422: "donnees_invalides", 500: "erreur_interne"}


class Probleme(Exception):
    """Une erreur métier qui doit sortir telle quelle, avec son code stable."""

    def __init__(self, status: int, code: str, detail: str) -> None:
        super().__init__(detail)
        self.status, self.code, self.detail = status, code, detail


def reponse(request: Request, status: int, code: str, detail: str,
            errors: list[dict[str, str]] | None = None) -> JSONResponse:
    corps: dict[str, Any] = {
        "type": TYPE_DE_BASE + code,
        "title": TITRES.get(status, HTTPStatus(status).phrase),
        "status": status,
        "detail": detail,
        "instance": request.state.request_id,
        "code": code,
    }
    if errors:
        corps["errors"] = errors
    return JSONResponse(corps, status_code=status, media_type=MEDIA,
                        headers={"Cache-Control": "no-store"})


def installe(app: FastAPI) -> None:
    @app.exception_handler(Probleme)
    async def _probleme(request: Request, exc: Probleme) -> JSONResponse:
        return reponse(request, exc.status, exc.code, exc.detail)

    @app.exception_handler(StarletteHTTPException)
    async def _http(request: Request, exc: StarletteHTTPException) -> JSONResponse:
        code = CODES.get(exc.status_code, "erreur")
        detail = "Cette adresse n'existe pas." if exc.status_code == 404 else TITRES.get(exc.status_code, "Erreur")
        return reponse(request, exc.status_code, code, detail)

    @app.exception_handler(RequestValidationError)
    async def _validation(request: Request, exc: RequestValidationError) -> JSONResponse:
        erreurs = [{"pointer": "/" + "/".join(str(p) for p in e["loc"][1:]), "message": e["msg"]}
                   for e in exc.errors()]
        return reponse(request, 422, "donnees_invalides", "Certaines données ne sont pas valides.", erreurs)

    @app.exception_handler(Exception)
    async def _inattendue(request: Request, exc: Exception) -> JSONResponse:
        return reponse(request, 500, "erreur_interne", "Une erreur inattendue est survenue.")
```

```python
from enum import StrEnum

from pydantic import SecretStr
from pydantic_settings import BaseSettings, SettingsConfigDict


class Env(StrEnum):
    local = "local"
    essai = "essai"
    prod = "prod"


class Settings(BaseSettings):
    model_config = SettingsConfigDict(env_prefix="POPOTE_", extra="ignore")

    env: Env = Env.local
    database_url: SecretStr | None = None
    relay_secret: SecretStr | None = None

    @property
    def relais_exige(self) -> bool:
        return self.env is not Env.local
```

```python
# src/popote_api/main.py
"""La fabrique de l'application."""
from collections.abc import Awaitable, Callable
from uuid import uuid4

from fastapi import FastAPI, Request, Response

from popote_api import errors
from popote_api.config import Settings


def create_app(settings: Settings | None = None) -> FastAPI:
    settings = settings or Settings()
    app = FastAPI(title="Popote", version="0.1.0", docs_url=None, redoc_url=None,
                  openapi_url="/api/openapi.json")
    app.state.settings = settings

    @app.middleware("http")
    async def identifiant(request: Request, suite: Callable[[Request], Awaitable[Response]]) -> Response:
        request.state.request_id = uuid4().hex
        reponse = await suite(request)
        reponse.headers["X-Request-Id"] = request.state.request_id
        return reponse

    errors.installe(app)
    return app


app = create_app()
```

```python
# src/popote_api/relais.py
"""Seul le relais Cloudflare peut appeler /api hors du poste local (spec § 2)."""
import hmac
from collections.abc import Awaitable, Callable

from fastapi import Request, Response

from popote_api.config import Settings
from popote_api.errors import reponse

EN_TETE = "x-popote-relais"
EN_TETE_IP = "x-popote-ip"


def relais_valide(request: Request, settings: Settings) -> bool:
    if settings.relay_secret is None:
        return False
    recu = request.headers.get(EN_TETE, "")
    return hmac.compare_digest(recu.encode(), settings.relay_secret.get_secret_value().encode())


def ip_de(request: Request) -> str:
    """L'adresse de la personne : celle que le relais a vue, seulement si le relais est authentique."""
    settings: Settings = request.app.state.settings
    if relais_valide(request, settings) and request.headers.get(EN_TETE_IP):
        return request.headers[EN_TETE_IP]
    return request.client.host if request.client else ""


async def garde(request: Request, suite: Callable[[Request], Awaitable[Response]]) -> Response:
    settings: Settings = request.app.state.settings
    if settings.relais_exige and request.url.path.startswith("/api/") and not relais_valide(request, settings):
        return reponse(request, 404, "introuvable", "Cette adresse n'existe pas.")
    return await suite(request)
```

```ts
// relais/src/index.ts
/**
 * Le relais de notre-popote.fr (spec § 2) : sert les fichiers du front avec les en-têtes de sécurité, et
 * transmet /api/* au conteneur de l'API avec l'en-tête secret et l'adresse IP vue par Cloudflare.
 */
export interface Env {
  ASSETS: Fetcher
  API_ORIGIN: string
  RELAY_SECRET: string
  /** `report-only` tant que le front parle encore à Supabase en direct (jusqu'au jour J). */
  CSP_MODE: 'enforce' | 'report-only'
  CSP: string
}

const SECURITE: Record<string, string> = {
  'Strict-Transport-Security': 'max-age=63072000; includeSubDomains; preload',
  'X-Content-Type-Options': 'nosniff',
  'Referrer-Policy': 'strict-origin-when-cross-origin',
  'Permissions-Policy': 'camera=(self), microphone=(), geolocation=(), payment=()',
  'Cross-Origin-Opener-Policy': 'same-origin',
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url)
    if (url.pathname.startsWith('/api/')) return relaie(request, url, env)
    const fichier = await env.ASSETS.fetch(request)
    const reponse = new Response(fichier.body, fichier)
    for (const [k, v] of Object.entries(SECURITE)) reponse.headers.set(k, v)
    reponse.headers.set(env.CSP_MODE === 'enforce' ? 'Content-Security-Policy'
      : 'Content-Security-Policy-Report-Only', env.CSP)
    return reponse
  },
}

async function relaie(request: Request, url: URL, env: Env): Promise<Response> {
  const cible = new URL(url.pathname + url.search, env.API_ORIGIN)
  const entetes = new Headers(request.headers)
  entetes.delete('host')
  entetes.set('X-Popote-Relais', env.RELAY_SECRET)
  entetes.set('X-Popote-IP', request.headers.get('CF-Connecting-IP') ?? '')
  const amont = await fetch(cible, {
    method: request.method,
    headers: entetes,
    body: request.method === 'GET' || request.method === 'HEAD' ? undefined : request.body,
    redirect: 'manual',
  })
  const reponse = new Response(amont.body, amont)
  if (!reponse.headers.has('Cache-Control')) reponse.headers.set('Cache-Control', 'no-store')
  return reponse
}
```

- [ ] **Step 1 : le script**

```bash
#!/usr/bin/env bash
# Sauvegarde chiffrée de la prod (spec § 5.4). Lit : PGHOST PGPORT PGUSER PGPASSWORD PGDATABASE,
# SUPABASE_URL SUPABASE_SECRET_KEY (fichiers des seaux), S3_* (Scaleway). Écrit dans le seau, rien en clair.
set -euo pipefail
jour=$(date -u +%Y-%m-%dT%H%MZ)
travail=$(mktemp -d); trap 'rm -rf "$travail"' EXIT
pg_dump --format=custom --no-owner --schema=public --schema=auth --schema=storage -f "$travail/base.dump"
pg_dumpall --roles-only --no-role-passwords -f "$travail/roles.sql" || echo "roles: partiel" > "$travail/roles.note"
psql -Atc "select table_schema||'.'||table_name, (xpath('/row/c/text()', query_to_xml('select count(*) as c from '||quote_ident(table_schema)||'.'||quote_ident(table_name), false, true, '')))[1]::text
           from information_schema.tables where table_schema in ('public','auth','storage') and table_type='BASE TABLE'" > "$travail/comptes.txt"
node ops/sauvegarde/copie-seaux.mjs "$travail/seaux"          # télécharge chaque objet des seaux
tar -C "$travail" -cf - . | age -R ops/sauvegarde/destinataires.txt > "$travail/popote-$jour.tar.age"
aws s3 cp "$travail/popote-$jour.tar.age" "s3://popote-sauvegardes/prod/popote-$jour.tar.age" \
  --endpoint-url https://s3.fr-par.scw.cloud
echo "sauvegarde: popote-$jour.tar.age ($(stat -c %s "$travail/popote-$jour.tar.age") octets)"
```
