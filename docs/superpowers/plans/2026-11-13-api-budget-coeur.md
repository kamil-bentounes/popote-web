create function public.budget_arrondi_des_parts() returns bigint
language sql immutable parallel safe set search_path = pg_catalog as $$ select 100::bigint $$;

-- Répartit un montant (en centimes) selon des poids, chaque part arrondie à l'euro : 'haut' (supérieur) ou 'bas' (inférieur).
create function public.budget_repartit(p_total_cents public.budget_centimes, p_poids bigint[], p_sens text)
returns bigint[] language plpgsql immutable parallel safe set search_path = pg_catalog as $$
declare
  v_pas constant bigint := public.budget_arrondi_des_parts();
  v_somme bigint; v_res bigint[] := '{}'; i int;
begin
  if p_total_cents < 0 or p_total_cents > 100000000 then
    raise exception 'total_hors_bornes' using errcode = '22003';
  end if;
  if p_sens not in ('haut', 'bas') or p_poids is null or cardinality(p_poids) not between 1 and 2 then
    raise exception 'arguments_invalides' using errcode = '22023';
  end if;
  select coalesce(sum(w), 0) into v_somme from unnest(p_poids) as w;
  if v_somme <= 0 or exists (select 1 from unnest(p_poids) as w where w < 0 or w > 100000000) then
    raise exception 'poids_invalides' using errcode = '22023';
  end if;
  for i in 1 .. cardinality(p_poids) loop
    v_res := v_res || case p_sens
      when 'haut' then ((p_total_cents * p_poids[i] + v_somme * v_pas - 1) / (v_somme * v_pas)) * v_pas
      else (p_total_cents * p_poids[i] / (v_somme * v_pas)) * v_pas end;
  end loop;
  return v_res;
end $$;

-- Le montant par mois d'une charge (§ 4 v4) : au centime SUPÉRIEUR, au profit du compte joint ; l'écart va au coussin. Période : month, quarter, year.
create function public.budget_mensualise(p_montant_cents public.budget_centimes, p_periode public.budget_periode) returns bigint
language sql immutable parallel safe set search_path = pg_catalog as $$
  select (p_montant_cents + n - 1) / n
    from (select case p_periode when 'month' then 1 when 'quarter' then 3 when 'year' then 12 end::bigint as n) t
   where n is not null and p_montant_cents >= 0
$$;

-- Le gardé d'une charge qui ne tombe pas chaque mois (D20, § 4 v4) : r = (mois − échéance) mod n ; mensualité × r sans jamais dépasser le montant ;
-- le mois de l'échéance : le montant entier. Il retombe à 0 quand l'opération est vue : c'est `budget_reserve` qui le dit.
create function public.budget_garde_du_mois(p_montant_cents public.budget_centimes, p_periode public.budget_periode, p_mois_d_echeance int, p_mois int)
returns bigint language sql immutable parallel safe set search_path = pg_catalog as $$
  with t as (select case p_periode when 'quarter' then 3 when 'year' then 12 end::int as n),
       r as (select t.n, (((p_mois - p_mois_d_echeance) % t.n) + t.n) % t.n as r from t where t.n is not null)
  select coalesce((select case when r.r = 0 then p_montant_cents
                               else least(p_montant_cents, public.budget_mensualise(p_montant_cents, p_periode) * r.r) end from r), 0) $$;

-- La réserve d'une charge fixe : UNE seule, jamais deux (§ 4 v4, tableau « Ce qui est réservé »).
--   `p_passee` est vrai si l'opération est vue (validée ou estimée), si la charge est avancée, **ou si elle est EN ATTENTE chez la banque** (déjà retranchée comme opération
--   en attente : arbitrage 3). mensuelle : son montant tant qu'elle n'est pas passée, 0 ensuite ; non mensuelle : son gardé, 0 si elle est passée ce mois-ci.
create function public.budget_reserve(p_montant_cents public.budget_centimes, p_periode public.budget_periode, p_mois_d_echeance int, p_mois int, p_passee boolean)
returns bigint language sql immutable parallel safe set search_path = pg_catalog as $$
  select case when p_passee then 0
              when p_periode = 'month' then p_montant_cents
              else public.budget_garde_du_mois(p_montant_cents, p_periode, p_mois_d_echeance, p_mois) end $$;

-- Ce qui manquera à l'échéance d'une charge posée en cours de cycle (D20) : ajouté aux virements de ce mois-là, selon le partage.
create function public.budget_manque_d_echeance(p_montant_cents public.budget_centimes, p_periode public.budget_periode, p_premier_mois date, p_mois_d_echeance date)
returns bigint language sql immutable parallel safe set search_path = pg_catalog as $$
  with t as (select case p_periode when 'quarter' then 3 else 12 end::int as n),
       c as (select t.n, least(t.n, greatest(0, (extract(year from p_mois_d_echeance) * 12 + extract(month from p_mois_d_echeance))::int
                                                 - (extract(year from p_premier_mois) * 12 + extract(month from p_premier_mois))::int + 1)) as c from t)
  select p_montant_cents - least(p_montant_cents, public.budget_mensualise(p_montant_cents, p_periode) * c.c) from c $$;

-- Le surplus d'arrondi du mois (D23) : ce que les parts arrondies à l'euro apportent de plus que le total exact.
create function public.budget_surplus_d_arrondi(p_parts_cents bigint[], p_total_cents public.budget_centimes) returns bigint
language sql immutable parallel safe set search_path = pg_catalog as $$
  select coalesce((select sum(p) from unnest(p_parts_cents) as p), 0) - p_total_cents $$;

-- « Pour finir le mois » et « par jour » (§ 4 v4) : solde comptable, moins ce qui est en attente chez la banque (une fois), moins ce qui est réservé, moins le coussin ;
-- euro inférieur, y compris pour un montant négatif ; jamais de « par jour » en dessous de 0.
create function public.budget_plancher_euro(p_cents bigint) returns bigint
language sql immutable parallel safe set search_path = pg_catalog as $$ select p_cents - (((p_cents % 100) + 100) % 100) $$;

create function public.budget_finir_le_mois(p_solde_cents public.budget_centimes, p_en_attente_cents public.budget_centimes, p_reserve_cents public.budget_centimes,
                                             p_coussin_cents public.budget_centimes, p_jours_restants int)
returns table (amount_cents bigint, per_day_cents bigint) language sql immutable parallel safe set search_path = pg_catalog as $$
  with m as (select public.budget_plancher_euro(p_solde_cents - p_en_attente_cents - p_reserve_cents - p_coussin_cents) as montant)
  select m.montant, case when m.montant >= 0 and p_jours_restants >= 1 then ((m.montant / p_jours_restants) / 100) * 100 end from m
$$;

-- Sans banque (§ 4 v4 et arbitrage 2) : les virements ANNONCÉS du mois, plus le gardé et le coussin REPORTÉS des mois passés, moins les dépenses notées.
create function public.budget_solde_sans_banque(p_annonces_cents public.budget_centimes, p_report_garde_cents public.budget_centimes,
                                                 p_report_coussin_cents public.budget_centimes, p_depenses_notees_cents public.budget_centimes)
returns bigint language sql immutable parallel safe set search_path = pg_catalog as $$
  select p_annonces_cents + p_report_garde_cents + p_report_coussin_cents - p_depenses_notees_cents $$;

-- Au point du mois (arbitrage 4) : une charge qui ne tombe pas chaque mois compte pour sa MENSUALITÉ, pas pour son débit ;
-- une charge mensuelle compte pour son débit réel.
create function public.budget_compte_au_point(p_montant_cents public.budget_centimes, p_periode public.budget_periode, p_debit_cents public.budget_centimes)
returns bigint language sql immutable parallel safe set search_path = pg_catalog as $$
  select case when p_periode = 'month' then p_debit_cents else public.budget_mensualise(p_montant_cents, p_periode) end $$;

-- Jours qui restent, aujourd'hui compris.
create function public.budget_jours_restants(p_jour date) returns int
language sql immutable parallel safe set search_path = pg_catalog as $$
  select ((date_trunc('month', p_jour::timestamp) + interval '1 month')::date - p_jour)::int $$;

-- Le virement du mois (§ 4) : jamais sous 0 € ; le reste de l'avance est reporté. Centimes admis.
create function public.budget_virement(p_part_cents public.budget_centimes, p_ajustement_cents public.budget_centimes, p_manque_cents public.budget_centimes, p_avances_cents public.budget_centimes, p_report_cents public.budget_centimes)
returns table (amount_cents bigint, carry_out_cents bigint) language sql immutable parallel safe set search_path = pg_catalog as $$
  select greatest(0, p_part_cents + p_ajustement_cents + p_manque_cents - p_avances_cents - p_report_cents),
         greatest(0, p_avances_cents + p_report_cents - (p_part_cents + p_ajustement_cents + p_manque_cents))
$$;

-- Le reste ou le manque du point (§ 10) : le reste à l'euro inférieur, le manque à l'euro supérieur ; l'écart d'arrondi va au coussin, pas au reste.
create function public.budget_point(p_prevu_cents public.budget_centimes, p_depense_cents public.budget_centimes)
returns table (leftover_cents bigint, shortfall_cents bigint) language sql immutable parallel safe set search_path = pg_catalog as $$
  select public.budget_plancher_euro(greatest(0, p_prevu_cents - p_depense_cents)),
         -public.budget_plancher_euro(-greatest(0, p_depense_cents - p_prevu_cents))
$$;

-- La part du salaire (§ 5.4) : pourcentage entier, demi vers le haut ; nul si le revenu est 0.
create function public.budget_effort_percent(p_part_cents public.budget_centimes, p_revenu_cents public.budget_centimes) returns int
language sql immutable parallel safe set search_path = pg_catalog as $$
  select case when p_revenu_cents > 0 then least(100000, (p_part_cents * 200 + p_revenu_cents) / (p_revenu_cents * 2))::int end $$;

-- Droits : fonctions pures, exécutables par les rôles d'application ; `public` et `anon` : rien (règle du dépôt, `ops/aligne_droits.py`).
do $$

end $$;
